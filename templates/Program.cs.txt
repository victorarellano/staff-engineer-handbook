using Level1.BackendFoundations.Examples;

namespace Level1.BackendFoundations;

internal static class Program
{
    private static readonly IReadOnlyDictionary<string, Example> Examples =
        new Dictionary<string, Example>
        {
            [Q001ValueTypesVsReferenceTypes.Number] = new(
                Q001ValueTypesVsReferenceTypes.Title,
                Q001ValueTypesVsReferenceTypes.Run)
        };

    private static int Main(string[] args)
    {
        if (args.Length == 0)
        {
            PrintAvailableExamples();
            return 0;
        }

        if (!Examples.TryGetValue(args[0], out var example))
        {
            Console.Error.WriteLine($"Example '{args[0]}' was not found.");
            Console.WriteLine();
            PrintAvailableExamples();
            return 1;
        }

        Console.WriteLine($"Question {args[0]} - {example.Title}");
        Console.WriteLine(new string('-', 60));
        Console.WriteLine();

        example.Run();
        return 0;
    }

    private static void PrintAvailableExamples()
    {
        Console.WriteLine("Staff Engineer Handbook");
        Console.WriteLine();
        Console.WriteLine("Usage:");
        Console.WriteLine(
            "  dotnet run --project src/Level1.BackendFoundations -- <number>");
        Console.WriteLine();
        Console.WriteLine("Available examples:");

        foreach (var item in Examples.OrderBy(item => item.Key))
        {
            Console.WriteLine($"  {item.Key} - {item.Value.Title}");
        }
    }

    private sealed record Example(string Title, Action Run);
}
